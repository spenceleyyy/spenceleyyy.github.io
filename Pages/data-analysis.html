Cumulative Pokémon by Primary Type

An animated bar chart race showing how Pokémon types accumulate across generations

R ggplot2 gganimate
R Code 
p <- ggplot(ranked, aes(x = rank, y = Cum, fill = Type1, group = Type1)) +
  geom_col(width = 0.80, color = NA) +
  # left-aligned type names with matching colors
  geom_text(aes(y = -left_inset,
                label = ifelse(Cum > 0, type_label, ""),
                colour = Type1),
            hjust = 1, vjust = 0.5, size = 4.6, fontface = "bold",
            lineheight = 0.9, check_overlap = TRUE) +
  # right value labels
  geom_text(aes(y = Cum + right_inset,
                label = ifelse(Cum > 0, value_label, ""),
                colour = Type1),
            hjust = 0, vjust = 0.5, size = 4, fontface = "bold",
            check_overlap = TRUE) +
  scale_fill_manual(values = pokemon_colors, drop = FALSE) +
  scale_colour_manual(values = pokemon_colors, drop = FALSE) +
  coord_flip(clip = "off") +
  scale_x_reverse() +
  scale_y_continuous(
    limits = c(-left_inset, max_cum * 1.12),
    labels = comma,
    expand = c(0, 0)
  ) +
  labs(
    title = "Cumulative Pokémon by Primary Type",
    subtitle = "All types • Up to Generation {closest_state}",
    x = NULL, y = NULL,
    caption = "Counts accumulate across generations"
  ) +
  theme_minimal(base_size = 14) +
  theme(
    legend.position = "none",
    axis.text = element_blank(),
    axis.title = element_blank(),
    panel.grid.major.y = element_blank(),
    panel.grid.minor = element_blank(),
    panel.grid.major.x = element_line(color = "#e9eef2"),
    plot.background  = element_rect(fill = "white", color = NA),
    panel.background = element_rect(fill = "white",  color = NA),
    plot.title = element_text(face = "bold", size = 22),
    plot.subtitle = element_text(size = 14, colour = "#555555"),
    plot.caption = element_text(size = 10, colour = "#777777"),
    plot.margin = margin(16, 36, 16, 96)
  ) +
  transition_states(Generation, transition_length = 4, state_length = 1) +
  view_follow(fixed_y = TRUE) +
  ease_aes("cubic-in-out")

# --- Animate inline (HTML) ---
print(animate(p, nframes = 140, fps = 20, renderer = magick_renderer()))

Interactive MPG Scatterplot

Explore fuel efficiency with hoverable tooltips for each vehicle.

R ggplot2 ggiraph
R Code 
# Hover Plot - Makes the names column
mtcars.df = mtcars
mtcars.df$name = rownames(mtcars.df)

# Make the hover plot
mpg.plot = ggplot(mtcars.df, aes(x = disp, y = mpg)) +
  geom_point_interactive(aes(tooltip = name), size = 2) +
  labs(title = "Hover to Reveal Name",
       x = "Miles Per Gallon",
       y = "Disp") +
  theme_bw()

girafe(
  code = print(mpg.plot),
  width_svg = 7, height_svg = 4,
  options = list(
    opts_sizing(rescale = TRUE)
  )
)
