Cumulative Pokémon by Primary Type

An animated bar chart race showing how Pokémon types accumulate across generations

R ggplot2 gganimate
R Code 
p <- ggplot(ranked, aes(x = rank, y = Cum, fill = Type1, group = Type1)) +
  geom_col(width = 0.80, color = NA) +
  # left-aligned type names with matching colors
  geom_text(aes(y = -left_inset,
                label = ifelse(Cum > 0, type_label, ""),
                colour = Type1),
            hjust = 1, vjust = 0.5, size = 4.6, fontface = "bold",
            lineheight = 0.9, check_overlap = TRUE) +
  # right value labels
  geom_text(aes(y = Cum + right_inset,
                label = ifelse(Cum > 0, value_label, ""),
                colour = Type1),
            hjust = 0, vjust = 0.5, size = 4, fontface = "bold",
            check_overlap = TRUE) +
  scale_fill_manual(values = pokemon_colors, drop = FALSE) +
  scale_colour_manual(values = pokemon_colors, drop = FALSE) +
  coord_flip(clip = "off") +
  scale_x_reverse() +
  scale_y_continuous(
    limits = c(-left_inset, max_cum * 1.12),
    labels = comma,
    expand = c(0, 0)
  ) +
  labs(
    title = "Cumulative Pokémon by Primary Type",
    subtitle = "All types • Up to Generation {closest_state}",
    x = NULL, y = NULL,
    caption = "Counts accumulate across generations"
  ) +
  theme_minimal(base_size = 14) +
  theme(
    legend.position = "none",
    axis.text = element_blank(),
    axis.title = element_blank(),
    panel.grid.major.y = element_blank(),
    panel.grid.minor = element_blank(),
    panel.grid.major.x = element_line(color = "#e9eef2"),
    plot.background  = element_rect(fill = "white", color = NA),
    panel.background = element_rect(fill = "white",  color = NA),
    plot.title = element_text(face = "bold", size = 22),
    plot.subtitle = element_text(size = 14, colour = "#555555"),
    plot.caption = element_text(size = 10, colour = "#777777"),
    plot.margin = margin(16, 36, 16, 96)
  ) +
  transition_states(Generation, transition_length = 4, state_length = 1) +
  view_follow(fixed_y = TRUE) +
  ease_aes("cubic-in-out")

# --- Animate inline (HTML) ---
print(animate(p, nframes = 140, fps = 20, renderer = magick_renderer()))

Parallel Coordinates: Car Specs

Adapted from a brushing exercise — hover any line to trace one car's specs across every measured feature.

R ggplot2 ggiraph
R Code 
mtcars.df <- mtcars %>% rownames_to_column("name")
l.mtcars.df <- mtcars.df %>%
  pivot_longer(-name, names_to = "feature", values_to = "value") %>%
  group_by(feature) %>%
  mutate(s.value = as.numeric(scale(value))) %>%
  ungroup()

p <- ggplot(l.mtcars.df, aes(feature, s.value, group = name)) +
  geom_line_interactive(
    aes(tooltip = paste0(name, " (", round(value, 1), ")"),
        data_id = name),
    alpha = 0.45, linewidth = 0.7, color = "#47304F") +
  theme_bw(base_size = 13) +
  labs(x = NULL, y = "Scaled value (z-score)",
       title = "Each line is a car — hover to trace one")

girafe(ggobj = p, width_svg = 9, height_svg = 5,
  options = list(
    opts_hover(css = "stroke:#e890be;stroke-width:3px;"),
    opts_hover_inv(css = "opacity:0.12;")
  ))

Gapminder: Life Expectancy vs. GDP

Adapted from a brushed Gapminder demo — animated across five decades, tracing the U.S. and Rwanda as economies and life expectancy diverge and recover.

R gganimate gapminder
R Code 
trace_countries <- c("United States", "Rwanda")

p <- ggplot(gapminder, aes(gdpPercap, lifeExp, size = pop, color = continent)) +
  geom_path(data = gapminder %>% filter(country %in% trace_countries),
            aes(group = country), color = "grey35", alpha = 0.7) +
  geom_point(alpha = 0.85) +
  geom_text(data = gapminder %>% filter(country %in% trace_countries, year == max(year)),
            aes(label = country), vjust = -1.3, fontface = "bold") +
  scale_x_log10(labels = comma) +
  scale_size(range = c(2, 16), guide = "none") +
  coord_cartesian(xlim = c(230, 60000), ylim = c(20, 85)) +
  labs(title = "Life expectancy vs. GDP per capita — {round(frame_time)}",
       x = "GDP per capita ($, log scale)", y = "Life expectancy (years)") +
  theme_bw(base_size = 15) +
  transition_time(year) + ease_aes("linear")

animate(p, nframes = 130, fps = 14,
        renderer = av_renderer("gapminder_bubble.mp4"))

UMAP Projection of Car Specs

Adapted from a parallel-coordinates + UMAP sensitivity demo — a single tuned embedding groups cars by engine size; hover a point for details.

R uwot ggiraph
R Code 
set.seed(42)
umap_out <- umap(mtcars.df %>% select(-name) %>% scale(),
                  n_neighbors = 8, min_dist = 0.05)
umap.df <- mtcars.df %>% mutate(UMAP1 = umap_out[,1], UMAP2 = umap_out[,2])

p <- ggplot(umap.df, aes(UMAP1, UMAP2, color = factor(cyl))) +

  theme_bw(base_size = 13) +
  labs(title = "UMAP projection of car specs")

Median Salary by Major Category

Adapted from a percentile-trimming Shiny exercise — violin + jitter plot of starting salaries by field; hover a point for the specific major.

R ggplot2 ggiraph
R Code 
majors_filtered <- grads %>% group_by(Major_category) %>% filter(n() >= 8) %>% ungroup()
cat_summ <- majors_filtered %>% group_by(Major_category) %>%
  summarise(mean_salary = mean(Median, na.rm = TRUE)) %>%
  mutate(Major_category = fct_reorder(Major_category, mean_salary))
majors_plot <- majors_filtered %>%
  mutate(Major_category = factor(Major_category, levels = levels(cat_summ$Major_category)))

p <- ggplot(majors_plot, aes(Major_category, Median, fill = Major_category)) +
  geom_violin(alpha = 0.55, color = NA) +
  geom_point_interactive(
    aes(tooltip = paste0(Major, "\n$", comma(Median)), data_id = Major),
    position = position_jitter(width = 0.15), alpha = 0.65, size = 1.7) +
  geom_hline(yintercept = median(majors_plot$Median, na.rm = TRUE), linetype = "dashed") +
  coord_flip() +
  scale_y_continuous(labels = comma) +
  theme_bw(base_size = 12) +
  labs(x = NULL, y = "Median salary ($)")

girafe(ggobj = p, width_svg = 9, height_svg = 6.5,
  options = list(opts_hover(css = "stroke:black;stroke-width:1.5px;r:4px;")))

Pick-a-Feature Scatterplot

Adapted from a dropdown-driven scatterplot demo — choose which spec to plot against MPG from the menu.

R plotly
R Code 
features <- setdiff(names(mtcars), "mpg")

buttons <- lapply(features, function(f) {
  list(method = "update",
       args = list(list(x = list(mtcars[[f]])),
                   list(xaxis = list(title = f))),
       label = f)
})

plot_ly(x = ~mtcars[[features[1]]], y = ~mtcars$mpg,
        type = "scatter", mode = "markers",
        text = rownames(mtcars), hoverinfo = "text+x+y") %>%
  layout(title = "Vehicle feature vs. MPG",
         updatemenus = list(list(buttons = buttons)))
